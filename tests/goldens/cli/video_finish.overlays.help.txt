usage: python -m gtm_core.video_finish overlays [-h] --shots SHOTS
                                                --ratio {16:9,1:1,4:5,9:16}
                                                --profile PROFILE
                                                [--product PRODUCT]
                                                [--profiles-root PROFILES_ROOT]
                                                [--shots-root SHOTS_ROOT]
                                                --out-dir OUT_DIR
                                                [--workdir WORKDIR]
                                                [--shot-ids SHOT_IDS]
                                                [--fps FPS]
                                                [--repo-root REPO_ROOT]
                                                [--content-root CONTENT_ROOT]
                                                [--json]

options:
  -h, --help            show this help message and exit
  --shots SHOTS         the shot list JSON; each shot's production.overlay is
                        its bubble spec
  --ratio {16:9,1:1,4:5,9:16}
  --profile PROFILE     active profile (brand-kit lookup)
  --product PRODUCT     active product slug (brand-kit lookup)
  --profiles-root PROFILES_ROOT
                        override profiles root for brand-kit lookup
  --shots-root SHOTS_ROOT
                        root each shot's 'file' is relative to (default: the
                        shot list's own folder)
  --out-dir OUT_DIR
  --workdir WORKDIR
  --shot-ids SHOT_IDS   comma-separated shot ids; omit for every shot carrying
                        a production.overlay
  --fps FPS             frame rate the bubble sequence is drawn at (default
                        30); the overlay filter matches it to the shot by
                        timestamp, so it need not equal the shot's own
  --repo-root REPO_ROOT
  --content-root CONTENT_ROOT
  --json
